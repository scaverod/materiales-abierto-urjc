\newcommand{\material}[3]{%
  \cleardoublepage
  \section{#1}
  \noindent #3\par
  \IfFileExists{#2}{%
    \includepdf[pages=-,pagecommand={\thispagestyle{plain}}]{#2}%
  }{%
    \begin{tcolorbox}[colback=urjcRed!5,colframe=urjcRed,title=Falta el fichero]
      No se encontró \texttt{\detokenize{#2}}. Cópialo en esa ruta
      (sin espacios ni tildes en el nombre: usa \texttt{tools/quitar\_tildes.py}).
    \end{tcolorbox}%
  }%
}
